document.getElementById('jobCreateForm').addEventListener('submit', async (e) => { e.preventDefault(); if (!requireRole(['Recruiter', 'Admin'], 'To create a job, you need Recruiter or Admin role.')) { return; } try { const body = { title: document.getElementById('jobTitle').value, description: document.getElementById('jobDescription').value, location: document.getElementById('jobLocation').value, salaryRange: document.getElementById('jobSalaryRange').value, requiredSkills: document.getElementById('jobRequiredSkills').value }; const job = await apiRequest('POST', '/jobs', body, true); showMessage('success', `Job created with id ${job.id}.`); loadJobs(); e.target.reset(); } catch (err) { showMessage('danger', 'Failed to create job: ' + err.message); } });
